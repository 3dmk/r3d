ate='DIRT-CUT';ai.targetLane=(Math.random()<.5?-1:1)*(9+ai.aggression*3)}
   else{ai.state='RACE';ai.targetLane=clamp(line+(Math.random()-.5)*.6,-4.7,4.7)}
  }

  const sev=cornerSeverity(near.idx),look=Math.floor(clamp(11+Math.abs(long)*.28-sev*10,8,28));
  let target=routeLookahead(near.idx,look,ai.lane);
  if(near.dist>16){ai.state='RECOVER';target=routeLookahead(near.idx,7,0)}

  for(const other of racers){
   if(!other||other===ai)continue;
   const d=ai.pos.distanceTo(other.pos);if(d<7&&d>.001){const away=ai.pos.clone().sub(other.pos).normalize();target.addScaledVector(away,(7-d)*.45)}
  }

        err=wrapAngle(desiredHeading-ai.heading),steer=clamp(err*(1.55+ai.aggression*.45),-1,1),

  if(lead&&leadDist<8)desiredSpeed=Math.min(desiredSpeed,Math.max(10,Math.abs(lead.speed||0))*1.02);
  const surf=terrainType(ai.pos);if(surf==='DIRT')desiredSpeed*=.82;else if(surf==='ROUGH')desiredSpeed*=.68;else if(surf==='SHOULDER')desiredSpeed*=.9;
  if(near.dist>15)desiredSpeed=Math.min(desiredSpeed,22);
  if(ai.health<45)desiredSpeed*=.9;

  const controls={
   throttle:long<desiredSpeed-1?1:long<desiredSpeed+1?.35:0,
   brake:long>desiredSpeed+2?clamp((long-desiredSpeed)/16,0,1):0,
   steer,
   handbrake:sevAhead>.48&&long>29?.3:0,
   nitro:ai.state==='OVERTAKE'&&leadDist<8&&ai.personality==='AGGRESSIVE'
  };
  const r=hybridMove(ai,controls,dt);long=r.lv.long;const lat=r.lv.lat;near=r.near;

  if(near.dist>27){const corr=near.p.clone().sub(ai.pos);corr.y=0;if(corr.lengthSq()>.001)ai.vel.addScaledVector(corr.normalize(),10*dt)}
  const prog=projectedProgress(ai.pos),dp=Math.abs(prog-(ai.lastProgress||prog));
  ai.stuck=(Math.abs(long)<1.5||dp<.00002)?ai.stuck+dt:Math.max(0,ai.stuck-dt*2);ai.lastProgress=prog;
  if(near.dist>45||ai.stuck>3.2){ai.pos.copy(near.p);ai.pos.y=sampleTrackSurface(ai.pos).y+.18;ai.heading=trackHeading(near.idx);ai.vel.set(0,0,0);ai.stuck=0;ai.state='RECOVER'}

  ai.speed=long;ai.progress=prog;ai.g.position.copy(ai.pos);ai.g.rotation.y=ai.heading;animateBuggy(ai,long,lat,dt);
  if(ai.progress>.27&&ai.checkpoint===0)ai.checkpoint=1;if(ai.progress>.58&&ai.checkpoint===1)ai.checkpoint=2;if(ai.progress<.06&&ai.checkpoint===2&&ai.speed>3){ai.lap=(ai.lap||0)+1;ai.checkpoint=0}
  if(!BOX3D.active){
   for(const o of destructibles)resolveWorldCollision(ai,o,2.0,true);
   for(const b of barriers)resolveWorldCollision(ai,b,1.9,true);
  }
  derivedActions(ai,controls,dt,false);
 }
 if(!BOX3D.active)for(let i=0;i<ais.length;i++)for(let j=i+1;j<ais.length;j++)resolveCarCollision(ais[i],ais[j]);
}
