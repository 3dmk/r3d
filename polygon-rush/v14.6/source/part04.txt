function box3dBuildTrackWorld(){
 if(!BOX3D.active||BOX3D.worldBuilt||!trackSamples?.length)return;
 box3dClearStatics();

 // Terrain floor: broad support under the whole track area.
 box3dCreateStaticBox(0,-1.2,0,120,1.0,120,0,100);

 // Approximate the road with short oriented static boxes.
 const step=4;
 for(let i=0;i<trackSamples.length;i+=step){
  const a=trackSamples[i],b=trackSamples[(i+step)%trackSamples.length];
  const mx=(a.x+b.x)*.5,my=(a.y+b.y)*.5-.18,mz=(a.z+b.z)*.5;
  const dx=b.x-a.x,dz=b.z-a.z,len=Math.hypot(dx,dz);
  const yaw=Math.atan2(dx,dz);
  box3dCreateStaticBox(mx,my,mz,4.9,.18,Math.max(1.2,len*.52),yaw,101);
 }

 // Existing breakable barriers become Box3D static colliders.
 for(const b of barriers||[]){
  if(!b?.position)continue;
  const sc=b.scale||{x:1,y:1,z:1};
  box3dCreateStaticBox(b.position.x,b.position.y,b.position.z,
    Math.max(.45,sc.x*.65),Math.max(.45,sc.y*.65),Math.max(.45,sc.z*.65),
    b.rotation?.y||0,200);
 }

 // Existing destructible rocks/props get coarse rigid colliders.
 for(const o of destructibles||[]){
  if(!o?.position)continue;
  box3dCreateStaticBox(o.position.x,o.position.y,o.position.z,1.1,1.0,1.1,o.rotation?.y||0,201);
 }
 BOX3D.worldBuilt=true;
}
function box3dPollContacts(){
 if(!BOX3D.active||!BOX3D.module)return;
 BOX3D.contacts.length=0;
 const p=BOX3D.module._pr_b3_get_contacts(BOX3D.world);
 if(!p)return;
 const i32=BOX3D.module.HEAP32,f32=BOX3D.module.HEAPF32,base=p>>2,count=i32[base];
 let off=base+1;
 for(let i=0;i<count;i++){
  const bodyA=i32[off++],bodyB=i32[off++],tagA=i32[off++],tagB=i32[off++],
        impact=f32[off++],nx=f32[off++],ny=f32[off++],nz=f32[off++];
  BOX3D.contacts.push({bodyA,bodyB,tagA,tagB,impact,nx,ny,nz});BOX3D.contactHits++;BOX3D.lastImpact=Math.max(BOX3D.lastImpact,impact);
 }
 BOX3D.module._pr_b3_free_contacts(p);
}
function box3dHandleContacts(){
 if(!BOX3D.active||!BOX3D.contacts.length)return;
 const reverse=new Map();for(const [car,id] of BOX3D.bodies.entries())reverse.set(id,car);
 for(const c of BOX3D.contacts){
  const car=reverse.get(c.bodyA)||reverse.get(c.bodyB);
  if(!car)continue;
  const severity=clamp(c.impact/28,0,1.4);
  if(severity>.12)applyImpactDamage(car,severity,'collision');
  if(car===player&&severity>.18)cameraShake=Math.max(cameraShake,Math.min(.9,severity*.75));
 }
}

}
function box3dStep(dt){
 if(!BOX3D.active||!BOX3D.module)return;
 if(!BOX3D.worldBuilt)box3dBuildTrackWorld();
 BOX3D.accum=Math.min(.25,BOX3D.accum+dt);
 while(BOX3D.accum>=BOX3D.fixedDt){
  BOX3D.module._pr_b3_step(BOX3D.world,BOX3D.fixedDt,BOX3D.subSteps);
  BOX3D.accum-=BOX3D.fixedDt;
  box3dPollContacts();
  box3dHandleContacts();
 }
}

 return true;

function hybridMove(car,controls,dt){
 if(BOX3D.active){
  box3dApplyControl(car,controls);
  const ok=box3dSyncCar(car);
  if(ok){
   const near=nearestTrack(car.pos),surf=sampleTrackSurface(car.pos),lv=localVelocity(car);
   car.speed=lv.long;car.slip=Math.atan2(Math.abs(lv.lat),Math.abs(lv.long)+.25);
   return {surf,near,lv,surface:surf.type};
  }
 }
 return applyArcadeMovement(car,controls,dt);
}

function ensureMoveState(car){
 car.yawRate??=0;car.health??=100;car.nitro??=100;car.suspensionDamage??=0;
 car.engineDamage??=0;car.steeringDamage??=0;car.brakeDamage??=0;

}
function localVelocity(car){
 const c=Math.cos(car.heading),si=Math.sin(car.heading),vx=car.vel.x,vz=car.vel.z;
 return {lat:vx*c-vz*si,long:vx*si+vz*c};
}
function applyArcadeMovement(car,controls,dt){
 ensureMoveState(car);
 const before=localVelocity(car),speedAbs=Math.abs(before.long),surface=terrainType(car.pos),surfaceGrip=terrainGrip(surface);

 const steer=(controls.steer||0)*MOVE.steerRate*steerAuthority;

 if((controls.throttle||0)>0&&long<maxForward)long+=accel*(controls.throttle||0)*dt;
 if((controls.brake||0)>0){
   const b=MOVE.brake*brakeHealth*(controls.brake||0)*dt;
   if(long>0)long=Math.max(0,long-b);
   else long=Math.max(-MOVE.maxReverse,long-MOVE.reverse*dt);
 }
 if(!(controls.throttle||0)&&!(controls.brake||0))long*=Math.pow(surface==='ASPHALT'?MOVE.drag:MOVE.offroadDrag,dt*60);

 const grip=gripBase*surfaceGrip*(1-car.suspensionDamage*.22);
 lat*=Math.pow(clamp(1-grip*dt,0,.999),1);
 if((controls.handbrake||0)&&speedAbs>8)lat+=steer
