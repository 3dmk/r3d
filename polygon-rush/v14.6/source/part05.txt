*speedAbs*.13;

 const near=nearestTrack(car.pos),surf=sampleTrackSurface(car.pos),groundY=surf.y+.18;

 const crest=nextY-prevY;

 }

   if(car.pos.y<=groundY){
     const impact=Math.abs(car.vy);car.pos.y=groundY;car.airborne=false;car.airTime=0;
     if(impact>5.8){car.suspensionDamage=clamp(car.suspensionDamage+(impact-5.8)*.032,0,.8);car.health=Math.max(0,car.health-(impact-5.8)*1.4);car.vel.multiplyScalar(.94);cameraShake=Math.max(cameraShake,Math.min(.85,impact*.07))}
     car.vy=0;
   }
 }else{
   car.pos.y+=(groundY-car.pos.y)*Math.min(1,dt*10);car.vy=0;
 }

 car.pos.addScaledVector(car.vel,dt);
 const after=localVelocity(car);
 car.slip=Math.atan2(Math.abs(after.lat),Math.abs(after.long)+.25);
 car.speed=after.long;
 car.bottomOut=surface==='ROUGH'&&speedAbs>24&&Math.sin((car.pos.x+car.pos.z)*.18)>.86;

 const vis=car.g?.userData?.visual;
 if(vis){
   const terrainPitch=clamp((nextY-prevY)*.05,-.12,.12),dynPitch=clamp((after.long-before.long)*-.008,-.08,.08),
         dynRoll=clamp(-(controls.steer||0)*Math.min(speedAbs/30,1)*.12-after.lat*.008,-.17,.17);
   car.g.userData.pitch+=(terrainPitch+dynPitch-car.g.userData.pitch)*Math.min(1,dt*7);
   car.g.userData.roll+=(dynRoll-car.g.userData.roll)*Math.min(1,dt*7);
   vis.rotation.x=car.g.userData.pitch;vis.rotation.z=car.g.userData.roll;
 }
 return {surf,near,lv:after,surface};
}
function applyImpactDamage(car,severity,kind='collision'){
 ensureMoveState(car);

 if(kind==='landing'||kind==='bottom')car.suspensionDamage=clamp(car.suspensionDamage+d*.12,0,.8);
 else{
   car.engineDamage=clamp(car.engineDamage+d*.045,0,.7);
   car.steeringDamage=clamp(car.steeringDamage+d*.035,0,.65);
   car.brakeDamage=clamp(car.brakeDamage+d*.025,0,.6);
 }

}
function resolveCarCollision(a,b){
 const dvec=a.pos.clone().sub(b.pos);dvec.y=0;const d=dvec.length(),r=MOVE.collisionRadius;
 if(d<=.001||d>=r)return;
 const n=dvec.multiplyScalar(1/d),pen=r-d,rel=a.vel.clone().sub(b.vel).dot(n);
 a.pos.addScaledVector(n,pen*.5);b.pos.addScaledVector(n,-pen*.5);
 if(rel<0){
   const impulse=-rel*.58;a.vel.addScaledVector(n,impulse);b.vel.addScaledVector(n,-impulse);
   a.heading+=new THREE.Vector3(Math.cos(a.heading),0,-Math.sin(a.heading)).dot(n)*impulse*.018;
   b.heading-=new THREE.Vector3(Math.cos(b.heading),0,-Math.sin(b.heading)).dot(n)*impulse*.018;
   applyImpactDamage(a,Math.abs(rel)/22,'collision');applyImpactDamage(b,Math.abs(rel)/22,'collision');
 }
}
function resolveWorldCollision(car,obj,radius=1.9,breakable=false){
 if(!obj.visible)return;
 const dvec=car.pos.clone().sub(obj.position);dvec.y=0;const d=dvec.length();if(d<=.001||d>radius)return;
 const n=dvec.multiplyScalar(1/d),speed=car.vel.length();
 if(breakable&&speed>14){obj.visible=false;obj.userData.alive=false;car.vel.multiplyScalar(.9);applyImpactDamage(car,speed/48,'collision');return}
 car.pos.addScaledVector(n,radius-d+.02);const vn=car.vel.dot(n);if(vn<0)car.vel.addScaledVector(n,-vn*1.35);
 applyImpactDamage(car,speed/52,'collision');
}
